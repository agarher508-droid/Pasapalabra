          this.ctx.resume();
        }
      }

      playCorrect() {
        if (!this.enabled) return;
        this.init();
        if (!this.ctx) return;

        const now = this.ctx.currentTime;
        const osc1 = this.ctx.createOscillator();
        const osc2 = this.ctx.createOscillator();
        const gain = this.ctx.createGain();

        osc1.type = 'sine';
        osc2.type = 'triangle';

        // Arpeggio C5 -> E5 -> G5
        osc1.frequency.setValueAtTime(523.25, now);
        osc1.frequency.setValueAtTime(659.25, now + 0.08);
        osc1.frequency.setValueAtTime(783.99, now + 0.16);

        osc2.frequency.setValueAtTime(261.63, now);
        osc2.frequency.setValueAtTime(329.63, now + 0.08);
        osc2.frequency.setValueAtTime(392.00, now + 0.16);

        gain.gain.setValueAtTime(0.2, now);
        gain.gain.exponentialRampToValueAtTime(0.001, now + 0.4);

        osc1.connect(gain);
        osc2.connect(gain);
        gain.connect(this.ctx.destination);

        osc1.start(now);
        osc2.start(now);
        osc1.stop(now + 0.4);
        osc2.stop(now + 0.4);
      }

      playIncorrect() {
        if (!this.enabled) return;
        this.init();
        if (!this.ctx) return;

        const now = this.ctx.currentTime;
        const osc = this.ctx.createOscillator();
        const gain = this.ctx.createGain();

        osc.type = 'sawtooth';
        osc.frequency.setValueAtTime(180, now);
        osc.frequency.linearRampToValueAtTime(110, now + 0.3);

        gain.gain.setValueAtTime(0.2, now);
        gain.gain.exponentialRampToValueAtTime(0.01, now + 0.3);

        osc.connect(gain);
        gain.connect(this.ctx.destination);

        osc.start(now);
        osc.stop(now + 0.3);
      }

      playSkip() {
        if (!this.enabled) return;
        this.init();
        if (!this.ctx) return;

        const now = this.ctx.currentTime;
        const osc = this.ctx.createOscillator();
        const gain = this.ctx.createGain();

        osc.type = 'sine';
        osc.frequency.setValueAtTime(400, now);
        osc.frequency.exponentialRampToValueAtTime(600, now + 0.12);

        gain.gain.setValueAtTime(0.15, now);
        gain.gain.exponentialRampToValueAtTime(0.01, now + 0.12);

        osc.connect(gain);
        gain.connect(this.ctx.destination);

        osc.start(now);
        osc.stop(now + 0.12);
      }

      playVictory() {
        if (!this.enabled) return;
        this.init();
        if (!this.ctx) return;

        const now = this.ctx.currentTime;
        const notes = [523.25, 659.25, 783.99, 1046.50];
        notes.forEach((freq, i) => {
          const osc = this.ctx.createOscillator();
          const gain = this.ctx.createGain();
          osc.type = 'triangle';
          osc.frequency.value = freq;
          gain.gain.setValueAtTime(0.2, now + i * 0.12);
          gain.gain.exponentialRampToValueAtTime(0.001, now + i * 0.12 + 0.3);
          osc.connect(gain);
          gain.connect(this.ctx.destination);
          osc.start(now + i * 0.12);
          osc.stop(now + i * 0.12 + 0.3);
        });
      }
    }

    const soundSystem = new SoundEffects();

    function toggleAudio() {
      soundSystem.enabled = !soundSystem.enabled;
      const icon = document.getElementById('soundIcon');
      if (soundSystem.enabled) {
        icon.className = 'fas fa-volume-up text-amber-400';
      } else {
        icon.className = 'fas fa-volume-mute text-slate-500';
      }
    }

    /* ==========================================================================
       3. GAME STATE MANAGEMENT
       ========================================================================== */
    let gameMode = 'single'; // 'single' | 'team'
    let currentTeamTurn = 1; // 1 | 2
    let lettersState = []; // Array of { ...ROSCO_DATA[i], status: 'pending'|'correct'|'incorrect' }
    let currentLetterIndex = 0;
    let timerSeconds = 180; // 3 minutes standard
    let timerInterval = null;
    let isGameRunning = false;

    // Scores
    let singleScore = { correct: 0, incorrect: 0 };
    let teamScores = {
      team1: { name: 'Equipo Azul', correct: 0, incorrect: 0 },
      team2: { name: 'Equipo Rojo', correct: 0, incorrect: 0 }
    };

    /* ==========================================================================
       4. INITIALIZATION & SETUP FUNCTIONS
       ========================================================================== */
    function selectMode(mode) {
      gameMode = mode;
      const singleBtn = document.getElementById('modeSingleBtn');
      const teamBtn = document.getElementById('modeTeamBtn');
      const teamNamesContainer = document.getElementById('teamNamesContainer');

      if (mode === 'single') {
        singleBtn.className = 'py-3 rounded-xl font-bold transition flex items-center justify-center gap-2 bg-indigo-600 text-white shadow-md';
        teamBtn.className = 'py-3 rounded-xl font-bold transition flex items-center justify-center gap-2 text-slate-400 hover:text-white';
        teamNamesContainer.classList.add('hidden');
      } else {
        teamBtn.className = 'py-3 rounded-xl font-bold transition flex items-center justify-center gap-2 bg-indigo-600 text-white shadow-md';
        singleBtn.className = 'py-3 rounded-xl font-bold transition flex items-center justify-center gap-2 text-slate-400 hover:text-white';
        teamNamesContainer.classList.remove('hidden');
      }
    }

    function startGame() {
      soundSystem.init();

      // Read team names if team mode
      if (gameMode === 'team') {
        teamScores.team1.name = document.getElementById('team1Input').value.trim() || 'Equipo Azul';
        teamScores.team2.name = document.getElementById('team2Input').value.trim() || 'Equipo Rojo';
        document.getElementById('teamTurnIndicator').classList.remove('hidden');
      } else {
        document.getElementById('teamTurnIndicator').classList.add('hidden');
      }

      // Reset state variables
      lettersState = ROSCO_DATA.map(item => ({
        ...item,
        status: 'pending' // pending, correct, incorrect
      }));

      currentLetterIndex = 0;
      currentTeamTurn = 1;
      timerSeconds = 180;
      singleScore = { correct: 0, incorrect: 0 };
      teamScores.team1.correct = 0;
      teamScores.team1.incorrect = 0;
      teamScores.team2.correct = 0;
      teamScores.team2.incorrect = 0;

      // Hide modal & setup, show gameplay UI
      document.getElementById('setupScreen').classList.add('hidden');
      document.getElementById('summaryModal').classList.add('hidden');
      document.getElementById('gamePlayScreen').classList.remove('hidden');

      // Render Wheel Nodes & initial prompt
      renderRoscoWheel();
      updateStatsDisplay();
      updateTurnIndicator();

      // Start Timer
      clearInterval(timerInterval);
      isGameRunning = true;
      updateTimerUI();
      timerInterval = setInterval(tickTimer, 1000);

      // Focus input
      selectNextPendingLetter(0);
    }

    function renderRoscoWheel() {
      const container = document.getElementById('roscoWheel');
      container.innerHTML = '';

      const total = lettersState.length;
      // Calculate positions on circle
      const isMobile = window.innerWidth < 640;
      const radius = isMobile ? 128 : 175; // Distance from center
      const centerX = isMobile ? 160 : 220;
      const centerY = isMobile ? 160 : 220;

      lettersState.forEach((item, index) => {
        // Angle in radians (start from top = -90deg or -PI/2)
        const angle = (index / total) * (2 * Math.PI) - (Math.PI / 2);
        const x = centerX + radius * Math.cos(angle) - (isMobile ? 17 : 22);
        const y = centerY + radius * Math.sin(angle) - (isMobile ? 17 : 22);

        const node = document.createElement('div');
        node.id = `letter-node-${index}`;
        node.className = `letter-node letter-pending`;
        node.style.left = `${x}px`;
        node.style.top = `${y}px`;
        node.innerText = item.letter;

        container.appendChild(node);
      });
    }

    function updateRoscoNodesUI() {
      lettersState.forEach((item, index) => {
        const node = document.getElementById(`letter-node-${index}`);
        if (!node) return;

        // Base styling based on status
        if (item.status === 'correct') {
          node.className = 'letter-node letter-correct';
        } else if (item.status === 'incorrect') {
          node.className = 'letter-node letter-incorrect';
        } else {
          node.className = 'letter-node letter-pending';
        }

        // Active highlighted letter
        if (index === currentLetterIndex && isGameRunning) {
          node.classList.add('letter-active');
        }
      });
    }

    /* ==========================================================================
       5. GAMEPLAY CONTROLS & LOGIC
       ========================================================================== */
    function selectNextPendingLetter(startIndex = 0) {
      if (!hasPendingLetters()) {
        endGame();
        return;
      }

      let idx = startIndex;
      let count = 0;
      const total = lettersState.length;

      // Find next letter with 'pending' status
      while (lettersState[idx].status !== 'pending' && count < total) {
        idx = (idx + 1) % total;
        count++;
      }

      if (count >= total) {
        // No pending letters remain
        endGame();
        return;
      }

      currentLetterIndex = idx;
      updateRoscoNodesUI();
      displayCurrentPrompt();
    }

    function displayCurrentPrompt() {
      const current = lettersState[currentLetterIndex];
      if (!current) return;

      // Update Header Badge & Text
      document.getElementById('letterBadge').innerHTML = `
        <i class="fas fa-font"></i> ${current.prefix}
      `;
      document.getElementById('definitionText').innerText = current.prompt;

      // Reset Clue/Hint display
      document.getElementById('hintText').innerText = '💡 ¿Necesitas una pista?';
      document.getElementById('hintText').classList.add('opacity-80', 'italic');

      // Clear & focus input box
      const input = document.getElementById('answerInput');
      input.value = '';
      input.focus();
    }

    function revealHint() {
      const current = lettersState[currentLetterIndex];
      if (!current) return;
      const hintElem = document.getElementById('hintText');
      hintElem.innerText = `Traducción / Pista: ${current.hint}`;
      hintElem.classList.remove('opacity-80', 'italic');
    }

    function handleAnswerSubmit(e) {
      if (e) e.preventDefault();
      if (!isGameRunning) return;

      const inputElem = document.getElementById('answerInput');
      const userAnswer = normalizeText(inputElem.value);
      const currentItem = lettersState[currentLetterIndex];

      if (userAnswer === '') return; // Prevent empty submission

      // Validate against array of accepted answers or default word
      const acceptedAnswers = currentItem.accepted 
        ? currentItem.accepted.map(a => normalizeText(a)) 
        : [normalizeText(currentItem.word)];

      const isCorrect = acceptedAnswers.includes(userAnswer);

      if (isCorrect) {
        // CORRECT ANSWER
        lettersState[currentLetterIndex].status = 'correct';
        soundSystem.playCorrect();

        if (gameMode === 'team') {
          if (currentTeamTurn === 1) teamScores.team1.correct++;
          else teamScores.team2.correct++;
        } else {
          singleScore.correct++;
        }
      } else {
        // INCORRECT ANSWER
        lettersState[currentLetterIndex].status = 'incorrect';
        soundSystem.playIncorrect();

        // Shake effect on card
        const card = document.getElementById('definitionCard');
        card.classList.add('shake');
        setTimeout(() => card.classList.remove('shake'), 400);

        if (gameMode === 'team') {
          if (currentTeamTurn === 1) teamScores.team1.incorrect++;
          else teamScores.team2.incorrect++;
          // Switch turn in team mode on incorrect answer
          switchTeamTurn();
        } else {
          singleScore.incorrect++;
        }
      }

      updateStatsDisplay();
      selectNextPendingLetter((currentLetterIndex + 1) % lettersState.length);
    }

    function handlePasapalabra() {
      if (!isGameRunning) return;
      soundSystem.playSkip();

      if (gameMode === 'team') {
        switchTeamTurn();
      }

      // Move to next pending letter without changing status
      selectNextPendingLetter((currentLetterIndex + 1) % lettersState.length);
    }

    function handleGiveUpLetter() {
      if (!isGameRunning) return;
      // Mark current as incorrect automatically
      lettersState[currentLetterIndex].status = 'incorrect';
      soundSystem.playIncorrect();

      if (gameMode === 'team') {
        if (currentTeamTurn === 1) teamScores.team1.incorrect++;
        else teamScores.team2.incorrect++;
        switchTeamTurn();
      } else {
        singleScore.incorrect++;
      }

      updateStatsDisplay();
      selectNextPendingLetter((currentLetterIndex + 1) % lettersState.length);
    }

    function switchTeamTurn() {
      currentTeamTurn = currentTeamTurn === 1 ? 2 : 1;
      updateTurnIndicator();
    }

    function updateTurnIndicator() {
      if (gameMode !== 'team') return;
      const turnLabel = document.getElementById('currentTeamName');
      if (currentTeamTurn === 1) {
        turnLabel.innerText = teamScores.team1.name;
        turnLabel.className = 'text-amber-300 font-bold';
      } else {
        turnLabel.innerText = teamScores.team2.name;
        turnLabel.className = 'text-cyan-300 font-bold';
      }
    }

    function normalizeText(str) {
      return str
        .trim()
        .toLowerCase()
        .normalize("NFD")
        .replace(/[\u0300-\u036f]/g, "") // remove accents
        .replace(/[^a-z0-9 ]/g, ""); // remove special characters
    }

    function hasPendingLetters() {
      return lettersState.some(item => item.status === 'pending');
    }

    function updateStatsDisplay() {
      const correctCount = lettersState.filter(i => i.status === 'correct').length;
      const incorrectCount = lettersState.filter(i => i.status === 'incorrect').length;
      const pendingCount = lettersState.filter(i => i.status === 'pending').length;

      document.getElementById('statCorrect').innerText = correctCount;
      document.getElementById('statIncorrect').innerText = incorrectCount;
      document.getElementById('statPending').innerText = pendingCount;
    }

    /* ==========================================================================
       6. TIMER LOGIC
       ========================================================================== */
    function tickTimer() {
      if (!isGameRunning) return;

      timerSeconds--;
      updateTimerUI();

      if (timerSeconds <= 0) {
        clearInterval(timerInterval);
        endGame('timeout');
      }
    }

    function updateTimerUI() {
      const mins = Math.floor(timerSeconds / 60);
      const secs = timerSeconds % 60;
      const formatted = `${mins.toString().padStart(2, '0')}:${secs.toString().padStart(2, '0')}`;
      
      const display = document.getElementById('timerDisplay');
      display.innerText = formatted;

      if (timerSeconds <= 30) {
        display.classList.add('text-red-400');
        display.classList.remove('text-amber-300');
      } else {
        display.classList.add('text-amber-300');
        display.classList.remove('text-red-400');
      }
    }

    /* ==========================================================================
       7. END GAME & SUMMARY MODAL
       ========================================================================== */
    function endGame(reason = 'completed') {
      isGameRunning = false;
      clearInterval(timerInterval);
      soundSystem.playVictory();

      const modal = document.getElementById('summaryModal');
      const title = document.getElementById('modalTitle');
      const subtitle = document.getElementById('modalSubtitle');
      const singleStats = document.getElementById('singleSummaryStats');
      const teamStats = document.getElementById('teamSummaryStats');

      if (reason === 'timeout') {
        title.innerText = '¡Tiempo Agotado!';
        subtitle.innerText = 'Se ha terminado el tiempo del juego';
      } else {
        title.innerText = '¡Rosco Completado!';
        subtitle.innerText = '¡Has respondido a todas las letras!';
      }

      if (gameMode === 'single') {
        singleStats.classList.remove('hidden');
        teamStats.classList.add('hidden');

        const correct = singleScore.correct;
        const incorrect = singleScore.incorrect;
        const accuracy = (correct + incorrect) > 0 ? Math.round((correct / (correct + incorrect)) * 100) : 0;

        document.getElementById('modalCorrect').innerText = correct;
        document.getElementById('modalIncorrect').innerText = incorrect;
        document.getElementById('modalScore').innerText = `${accuracy}%`;
      } else {
        singleStats.classList.add('hidden');
        teamStats.classList.remove('hidden');

        document.getElementById('modalTeam1Name').innerText = teamScores.team1.name;
        document.getElementById('modalTeam1Correct').innerText = teamScores.team1.correct;

        document.getElementById('modalTeam2Name').innerText = teamScores.team2.name;
        document.getElementById('modalTeam2Correct').innerText = teamScores.team2.correct;
      }

      modal.classList.remove('hidden');
    }

    function confirmReset() {
      if (confirm('¿Quieres reiniciar la partida actual?')) {
        startGame();
      }
    }

    function resetToSetup() {
      clearInterval(timerInterval);
      isGameRunning = false;
      document.getElementById('summaryModal').classList.add('hidden');
      document.getElementById('gamePlayScreen').classList.add('hidden');
      document.getElementById('setupScreen').classList.remove('hidden');
    }

    // Re-calculate Rosco nodes on window resize for proper responsiveness
    window.addEventListener('resize', () => {
      if (isGameRunning) {
        renderRoscoWheel();
        updateRoscoNodesUI();
      }
    });
